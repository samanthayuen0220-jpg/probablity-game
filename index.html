    <!-- AUDIO & MATH GAME LOGIC -->

    <script>

        // --- NATIVE SOUND SYNTHESIZER ---

        let audioEnabled = true;

        let audioContext;

 

        function initAudio() {

            if (!audioContext) {

                audioContext = new (window.AudioContext || window.webkitAudioContext)();

            }

        }

 

        // Play procedural synthesizer sound effect

        function playSound(type) {

            if (!audioEnabled) return;

            try {

                initAudio();

                const ctx = audioContext;

                if (ctx.state === 'suspended') {

                    ctx.resume();

                }

 

                const osc = ctx.createOscillator();

                const gain = ctx.createGain();

                osc.connect(gain);

                gain.connect(ctx.destination);

 

                const now = ctx.currentTime;

 

                if (type === 'collision') {

                    // Soft organic sound for balls pop/re-arrange

                    osc.type = 'sine';

                    osc.frequency.setValueAtTime(200 + Math.random() * 80, now);

                    osc.frequency.exponentialRampToValueAtTime(80, now + 0.08);

                    gain.gain.setValueAtTime(0.04, now);

                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);

                    osc.start(now);

                    osc.stop(now + 0.08);

                } else if (type === 'success') {

                    // Arpeggio chord of success (major scale)

                    const notes = [261.63, 329.63, 392.00, 523.25]; // C4, E4, G4, C5

                    notes.forEach((freq, idx) => {

                        const noteOsc = ctx.createOscillator();

                        const noteGain = ctx.createGain();

                        noteOsc.connect(noteGain);

                        noteGain.connect(ctx.destination);

                       

                        noteOsc.type = 'triangle';

                        noteOsc.frequency.setValueAtTime(freq, now + idx * 0.08);

                       

                        noteGain.gain.setValueAtTime(0.12, now + idx * 0.08);

                        noteGain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.08 + 0.25);

                       

                        noteOsc.start(now + idx * 0.08);

                        noteOsc.stop(now + idx * 0.08 + 0.25);

                    });

                } else if (type === 'fail') {

                    // Descending buzz error chord

                    osc.type = 'sawtooth';

                    osc.frequency.setValueAtTime(180, now);

                    osc.frequency.linearRampToValueAtTime(110, now + 0.25);

                    gain.gain.setValueAtTime(0.12, now);

                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.28);

                    osc.start(now);

                    osc.stop(now + 0.28);

                } else if (type === 'card') {

                    // Card swipe/drawing sound

                    osc.type = 'sine';

                    osc.frequency.setValueAtTime(400, now);

                    osc.frequency.exponentialRampToValueAtTime(900, now + 0.15);

                    gain.gain.setValueAtTime(0.08, now);

                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.15);

                    osc.start(now);

                    osc.stop(now + 0.15);

                }

            } catch (e) {

                console.warn("Audio Context could not play sound automatically.", e);

            }

        }

 

        // Toggle Audio

        document.getElementById('btn-audio').addEventListener('click', () => {

            audioEnabled = !audioEnabled;

        // --- BOX CHAMBER SIMULATION (STATIC SCATTER LAYOUT) ---

        const canvas = document.getElementById('physics-canvas');

        const ctx = canvas.getContext('2d');

 

        // Ball Config Counts (Default starts at 2 each, max capped at 10)

        const counts = {

            red: 2,

            blue: 2,

            green: 2,

            yellow: 2

        };

 

        const colorMap = {

            red: { hex: '#f43f5e', shadow: 'rgba(244, 63, 94, 0.4)', text: 'Rose Red', bg: 'bg-rose-500/20 text-rose-300 border-rose-500/40' },

            blue: { hex: '#3b82f6', shadow: 'rgba(59, 130, 246, 0.4)', text: 'Ocean Blue', bg: 'bg-blue-500/20 text-blue-300 border-blue-500/40' },

            green: { hex: '#10b981', shadow: 'rgba(16, 185, 129, 0.4)', text: 'Emerald Green', bg: 'bg-emerald-500/20 text-emerald-300 border-emerald-500/40' },

            yellow: { hex: '#f59e0b', shadow: 'rgba(245, 158, 11, 0.4)', text: 'Golden Yellow', bg: 'bg-amber-500/20 text-amber-300 border-amber-500/40' }

        };

 

        let balls = [];

 

        // Box Bounding Dimensions (Resized dynamically based on total balls inside)

        let boxBounds = {

            x: 0,

            y: 0,

            width: 0,

            height: 0

        };

 

        // Recalculates physical box bounds inside canvas space depending on ball population count

        function updateBoxBounds() {

            const totalBalls = counts.red + counts.blue + counts.green + counts.yellow;

           

            // Dynamic scale factor based on ball counts (Minimum 35% scale, Maximum 90% scale)

            const minScale = 0.35;

            const maxScale = 0.90;

            const maxCapacity = 40; // 10 balls per color max

           

            const scale = minScale + (maxScale - minScale) * (Math.max(1, totalBalls) / maxCapacity);

           

            boxBounds.width = canvas.width * scale;

            boxBounds.height = canvas.height * scale;

            boxBounds.x = (canvas.width - boxBounds.width) / 2;

            boxBounds.y = (canvas.height - boxBounds.height) / 2;

        }

 

        class Ball {

            constructor(colorKey) {

                this.colorKey = colorKey;

                this.color = colorMap[colorKey].hex;

                this.glow = colorMap[colorKey].shadow;

                this.radius = 15;

                this.x = 0;

                this.y = 0;

            }

 

            // Scatter ball randomly without overlapping others and strictly inside the current dynamically-sized box boundary

            setRandomPosition(existingBalls) {

                let attempts = 0;

                let found = false;

                const margin = this.radius + 10;

 

                while (attempts < 300 && !found) {

                    const rx = boxBounds.x + margin + Math.random() * (boxBounds.width - margin * 2);

                    const ry = boxBounds.y + margin + Math.random() * (boxBounds.height - margin * 2);

                   

                    let overlap = false;

                    for (let other of existingBalls) {

                        const dist = Math.hypot(rx - other.x, ry - other.y);

                        // Ensure balls don't touch

                        if (dist < (this.radius + other.radius + 10)) {

                            overlap = true;

                            break;

                        }

                    }

 

                    if (!overlap) {

                        this.x = rx;

                        this.y = ry;

                        found = true;

                    }

                    attempts++;

                }

 

                // Safe fallback within bounds

                if (!found) {

                    this.x = boxBounds.x + margin + Math.random() * (boxBounds.width - margin * 2);

                    this.y = boxBounds.y + margin + Math.random() * (boxBounds.height - margin * 2);

                }

            }

 

            draw() {

                ctx.beginPath();

                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);

                ctx.fillStyle = this.color;

                ctx.shadowBlur = 8;

                ctx.shadowColor = this.glow;

                ctx.fill();

                ctx.shadowBlur = 0; // reset

                ctx.closePath();

 

                // Shiny specular reflection highlights

                ctx.beginPath();

                ctx.arc(this.x - this.radius/3, this.y - this.radius/3, this.radius/4, 0, Math.PI * 2);

                ctx.fillStyle = 'rgba(255, 255, 255, 0.45)';

                ctx.fill();

                ctx.closePath();

            }

        }

 

        // Setup Canvas Size Responsiveness

        function resizeCanvas() {

            const container = canvas.parentNode;

            canvas.width = container.clientWidth;

            canvas.height = container.clientWidth;

            syncBallsFromConfig(); // Re-center and layout on resize

        }

 

        // Rebuild and reposition static ball collection cleanly inside resized box

        function syncBallsFromConfig() {

            updateBoxBounds();

            const newBallArray = [];

           

            Object.keys(counts).forEach(colorKey => {

                const targetCount = counts[colorKey];

                for (let i = 0; i < targetCount; i++) {

                    const ball = new Ball(colorKey);

                    ball.setRandomPosition(newBallArray);

                    newBallArray.push(ball);

                }

            });

 

            balls = newBallArray;

            updateGlobalCountBadge();

            renderBox();

        }

 

        function updateGlobalCountBadge() {

            const total = counts.red + counts.blue + counts.green + counts.yellow;

            document.getElementById('ball-total-badge').textContent = `Total: ${total} Balls`;

            document.getElementById('total-count-label').textContent = total;

           

            const warningEl = document.getElementById('empty-warning');

            if (total === 0) {

                warningEl.classList.remove('hidden');

            } else {

                warningEl.classList.add('hidden');

            }

        }

 

        // Re-shuffle placement coordinates inside bounds

        function shuffleBox() {

            playSound('collision');

            updateBoxBounds();

            const rearranged = [];

            balls.forEach(b => {

                b.setRandomPosition(rearranged);

                rearranged.push(b);

            });

            balls = rearranged;

            renderBox();

            showToast("🌀 Layout shuffled inside the resized chamber!", "info");

        }

 

        // Render entire static container chamber with dynamic boundary box

        function renderBox() {

            ctx.clearRect(0, 0, canvas.width, canvas.height);

 

            // Subtle grid overlay

            ctx.strokeStyle = 'rgba(79, 70, 229, 0.08)';

            ctx.lineWidth = 1;

            for(let i = 0; i < canvas.width; i += 30) {

                ctx.beginPath();

                ctx.moveTo(i, 0);

                ctx.lineTo(i, canvas.height);

                ctx.stroke();

                ctx.beginPath();

                ctx.moveTo(0, i);

                ctx.lineTo(canvas.width, i);

                ctx.stroke();

            }

 

            // Draw the Dynamic Chamber border box centered in canvas

            ctx.shadowBlur = 15;

            ctx.shadowColor = 'rgba(99, 102, 241, 0.3)';

            ctx.strokeStyle = 'rgba(99, 102, 241, 0.6)';

            ctx.lineWidth = 3;

           

            // Draw rounded rectangle container

            const radius = 16;

            ctx.beginPath();

            ctx.moveTo(boxBounds.x + radius, boxBounds.y);

            ctx.lineTo(boxBounds.x + boxBounds.width - radius, boxBounds.y);

            ctx.quadraticCurveTo(boxBounds.x + boxBounds.width, boxBounds.y, boxBounds.x + boxBounds.width, boxBounds.y + radius);

            ctx.lineTo(boxBounds.x + boxBounds.width, boxBounds.y + boxBounds.height - radius);

            ctx.quadraticCurveTo(boxBounds.x + boxBounds.width, boxBounds.y + boxBounds.height, boxBounds.x + boxBounds.width - radius, boxBounds.y + boxBounds.height);

            ctx.lineTo(boxBounds.x + radius, boxBounds.y + boxBounds.height);

            ctx.quadraticCurveTo(boxBounds.x, boxBounds.y + boxBounds.height, boxBounds.x, boxBounds.y + boxBounds.height - radius);

            ctx.lineTo(boxBounds.x, boxBounds.y + radius);

            ctx.quadraticCurveTo(boxBounds.x, boxBounds.y, boxBounds.x + radius, boxBounds.y);

            ctx.closePath();

            ctx.fillStyle = 'rgba(15, 23, 42, 0.6)';

            ctx.fill();

            ctx.stroke();

 

            // Clear shadow settings for drawing balls

            ctx.shadowBlur = 0;

 

            // Draw all the static balls inside the container box

            balls.forEach(b => b.draw());

        }

        // Euclidean Algorithm for GCD

        function getGreatestCommonDivisor(a, b) {

            return b ? getGreatestCommonDivisor(b, a % b) : a;

        }

        function showToast(message, type = "success") {

            const container = document.getElementById('toast-container');

            if (type === "success") {

                toast.className += " bg-emerald-950 border-emerald-500/30 text-emerald-300";

                toast.innerHTML = `🟢 ${message}`;

            } else if (type === "error") {

                toast.className += " bg-rose-950 border-rose-500/30 text-rose-300";

                toast.innerHTML = `🔴 ${message}`;

            } else {

                toast.className += " bg-slate-800 border-slate-700 text-slate-200";

                toast.innerHTML = `🔵 ${message}`;

            }

 

            container.appendChild(toast);

            // Trigger animation

            setTimeout(() => {

                toast.classList.remove('translate-y-2', 'opacity-0');

            }, 10);

 

            // Self-destruct sequence

            setTimeout(() => {

                toast.classList.add('translate-y-1', 'opacity-0');

                setTimeout(() => toast.remove(), 300);

            }, 3000);

        }

        // Handle mathematical verification of user input fraction

        function submitAnswer() {

            if (!currentChallenge) return;

 

            const userNumRaw = document.getElementById('input-numerator').value;

            const userDenRaw = document.getElementById('input-denominator').value;

 

            const userNum = parseInt(userNumRaw);

            const userDen = parseInt(userDenRaw);

 

            if (isNaN(userNum) || isNaN(userDen) || userDen === 0) {

                showToast("Please enter valid numerator and denominator values!", "error");

                playSound('fail');

                return;

            }

 

            if (userNum < 0 || userDen < 0) {

                showToast("Fractions cannot have negative numbers!", "error");

                playSound('fail');

                return;

            }

 

            totalAnswered++;

            document.getElementById('score-total').textContent = totalAnswered;

 

            // Math Calculation variables

            const actualNum = currentChallenge.targetCount;

            const actualDen = currentChallenge.totalCount;

           

            // STRICT VALIDATION: Numerator and denominator must match exact physical counts in the box

            const isCorrect = (userNum === actualNum && userDen === actualDen);

           

            // Check if user answer is mathematically equivalent but physically unmatched (e.g. 1/4 instead of 2/8)

            const isEquivalent = (userNum * actualDen === userDen * actualNum);

 

            // Compute the ideal simplified fraction details

            const gcdVal = getGreatestCommonDivisor(actualNum, actualDen);

            const simplifiedNum = actualNum / gcdVal;

            const simplifiedDen = actualDen / gcdVal;

            const isSimplifiable = gcdVal > 1 && actualNum !== 0;

 

            // Format whole numbers instead of fractions like 1/1 or 0/1

            let simplifiedStr = "";

            if (simplifiedDen === 1) {

                simplifiedStr = `${simplifiedNum}`; // e.g. "1" instead of "1/1", "0" instead of "0/1"

            } else {

                simplifiedStr = `${simplifiedNum}/${simplifiedDen}`;

            }

 

            // UI feedback transition

            document.getElementById('state-active').classList.add('hidden');

            const feedbackState = document.getElementById('state-feedback');

            feedbackState.classList.remove('hidden');

 

            // Inject metrics in the feedback pane

            document.getElementById('feedback-your-num').textContent = userNum;

            document.getElementById('feedback-your-den').textContent = userDen;

            document.getElementById('feedback-act-num').textContent = actualNum;

            document.getElementById('feedback-act-den').textContent = actualDen;

 

            const headlineEl = document.getElementById('feedback-headline');

            const iconContainer = document.getElementById('feedback-icon-container');

            const detailTextEl = document.getElementById('feedback-detail-text');

            const simpNoteEl = document.getElementById('feedback-simplification-note');

 

            if (isCorrect) {

                streak++;

                playSound('success');

                headlineEl.textContent = "Awesome Job! Correct!";

                headlineEl.className = "text-lg font-black tracking-tight text-emerald-400";

               

                iconContainer.textContent = "🎉";

                iconContainer.className = "w-16 h-16 rounded-full flex items-center justify-center text-4xl mb-3 shadow-lg bg-emerald-950 border border-emerald-500/40 text-emerald-400 animate-bounce";

 

                // Explanation message

                detailTextEl.innerHTML = `Correct! There are exactly <strong>${actualNum} ${colorMap[currentChallenge.targetColor].text}</strong> balls out of a total of <strong>${actualDen}</strong>.`;

               

                // Show how it simplifies

                if (isSimplifiable) {

                    simpNoteEl.innerHTML = `💡 Great job! This fraction simplifies down to: <strong>${simplifiedStr}</strong> (dividing both by ${gcdVal}).`;

                } else {

                    if (simplifiedStr === "1") {

                        simpNoteEl.innerHTML = `✨ This simplifies down to a whole number! The final answer is <strong>1</strong> (100% certainty).`;

                    } else if (simplifiedStr === "0") {

                        simpNoteEl.innerHTML = `✨ There are no balls of this color! The final answer is <strong>0</strong> (0% probability).`;

                    } else {

                        simpNoteEl.innerHTML = `✨ This fraction is already in its simplest irreducible form!`;

                    }

                }

 

            } else {

                streak = 0; // Break streak

                playSound('fail');

                headlineEl.textContent = "Oops! That's Not Correct.";

                headlineEl.className = "text-lg font-black tracking-tight text-rose-400";

 

                iconContainer.textContent = "❌";

                iconContainer.className = "w-16 h-16 rounded-full flex items-center justify-center text-4xl mb-3 shadow-lg bg-rose-950 border border-rose-500/40 text-rose-400";

 

                if (isEquivalent) {

                    // Specific explanation for matched values but unmatched counts

                    detailTextEl.innerHTML = `Almost! Your math is equivalent, but <strong>you must enter the exact physical counts</strong> first!`;

                    

                    if (isSimplifiable) {

                        simpNoteEl.innerHTML = `⚠️ The box has <strong>${actualNum}</strong> target balls out of <strong>${actualDen}</strong>. You must enter <strong>${actualNum}/${actualDen}</strong> first. (It simplifies to <strong>${simplifiedStr}</strong>).`;

                    } else {

                        simpNoteEl.innerHTML = `⚠️ The box has <strong>${actualNum}</strong> target balls out of <strong>${actualDen}</strong>. You must enter <strong>${actualNum}/${actualDen}</strong> first.`;

                    }

                } else {

                    // Completely incorrect fraction

                    detailTextEl.innerHTML = `The box currently contains exactly <strong>${actualNum} ${colorMap[currentChallenge.targetColor].text}</strong> balls with <strong>${actualDen} total</strong>.`;

                   

                    if (isSimplifiable) {

                        simpNoteEl.innerHTML = `📉 The correct probability is <strong>${actualNum}/${actualDen}</strong>, which simplifies down to <strong>${simplifiedStr}</strong>.`;

                    } else {

                        if (simplifiedStr === "1") {

                            simpNoteEl.innerHTML = `📉 The correct probability is <strong>${actualNum}/${actualDen}</strong>, which simplifies to <strong>1</strong>.`;

                        } else if (simplifiedStr === "0") {

                            simpNoteEl.innerHTML = `📉 The correct probability is <strong>${actualNum}/${actualDen}</strong>, which simplifies to <strong>0</strong>.`;

                        } else {

                            simpNoteEl.innerHTML = `📉 The correct probability is <strong>${actualNum}/${actualDen}</strong>.`;

                        }

                    }

                }

            }

 

            // Update streak metrics

            document.getElementById('score-streak').textContent = streak;

        }
